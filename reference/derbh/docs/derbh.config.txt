######################################
# General                            #
######################################
[Derbh]
autoload                        <string>                        Filename of archive to initialise on startup.
maxarchives                     <int>                           The limit of dz archives that can be attached (default is 32).
MinBufSize						<int>							Minimum buffer size to use when decompressing data. The default size is the compression window size used when compressing the data (this is the minimum). However larger buffers can be used, which facilitates seeking within a compressed file (seeking to data thats already buffered is free, whereas otherwise the file needs to be re-decompressed up to the seek point)

######################################
# Marmalade Studio trace channels      #
######################################
[Trace]
DERBH				<0 or 1>			If 1, output trace for Derbh module.
DERBH_VERBOSE		<0 or 1>			If 1, output detailed derbh trace, defaults to 0

######################################
# Marmalade Studio assertion channels  #
######################################
[Assert]
DERBH				<0 or 1>			If 1, output SysAssert for Derbh module.
